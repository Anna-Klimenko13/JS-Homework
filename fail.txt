import React, { useState, useEffect } from 'react';
import { Sparkles, Copy, Check, RefreshCw, ShoppingBag, History, Trash2, Eye, FileText, Star, ShoppingCart, Heart } from 'lucide-react';

export default function App() {
  const [productName, setProductName] = useState('');
  const [keywords, setKeywords] = useState('');
  const [tone, setTone] = useState('selling');
  const [marketplace, setMarketplace] = useState('amazon');
  const [price, setPrice] = useState('49.99');
  
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState(null);
  const [copied, setCopied] = useState(false);
  const [history, setHistory] = useState([]);
  
  const [activeTab, setActiveTab] = useState('text');

  useEffect(() => {
    const saved = localStorage.getItem('ai_cards_history');
    if (saved) {
      try { setHistory(JSON.parse(saved)); } catch (e) {}
    }
  }, []);

  const saveToHistory = (newItem) => {
    const updated = [newItem, ...history].slice(0, 3);
    setHistory(updated);
    localStorage.setItem('ai_cards_history', JSON.stringify(updated));
  };

  const clearHistory = () => {
    setHistory([]);
    localStorage.removeItem('ai_cards_history');
  };

  const generateSmartData = () => {
    const keysArray = keywords ? keywords.split(',').map(k => k.trim()).filter(Boolean) : ['premium quality', 'eco-friendly'];
    const mainKey = keysArray[0] || 'top rated';
    const secondKey = keysArray[1] || 'high durability';

    let titlePrefix = '';
    let descIntro = '';
    let bulletStyle = '';

    if (marketplace === 'amazon') titlePrefix = `[AMAZON CHOICE] `;
    if (marketplace === 'shopify') titlePrefix = `[TRENDING BRAND] `;
    if (marketplace === 'etsy') titlePrefix = `[UNIQUE & HANDMADE] `;

    if (tone === 'selling') {
      descIntro = `🔥 Максимум продаж! Представляем идеальный выбор для тех, кто ищет ${mainKey}.`;
      bulletStyle = 'Повышает комфорт с первых секунд использования';
    } else if (tone === 'professional') {
      descIntro = `Профессиональное качество. Модель разработана с акцентом на ${mainKey} и строгое соответствие международным стандартам.`;
      bulletStyle = 'Сертифицированные материалы и высокая износостойкость';
    } else {
      descIntro = `✨ Влюбитесь с первого взгляда! Этот ${productName || 'товар'} подарит вам невероятное ощущение уюта и стиля благодаря особенности "${mainKey}".`;
      bulletStyle = 'Идеально подходит в качестве подарка для близких';
    }

    return {
      id: Date.now(),
      productName: productName || 'Товар',
      marketplace: marketplace.toUpperCase(),
      price: price || '39.99',
      currency: marketplace === 'etsy' ? '€' : '$',
      title: `${titlePrefix}${productName || 'Товар'} / ${mainKey} / ${secondKey}`,
      description: `${descIntro} Наш ${productName || 'продукт'} обеспечивает идеальный баланс: ${keysArray.join(', ')}. Забудьте о компромиссах — выбирайте проверенное качество, которое уже оценили покупатели по всему миру!`,
      bullets: [
        `Главное преимущество: ${mainKey} на высшем уровне`,
        `Защитные технологии и долговечность (${secondKey})`,
        bulletStyle,
        'Быстрая и безопасная международная доставка'
      ]
    };
  };

  const handleGenerate = async (e) => {
    e.preventDefault();
    setLoading(true);
    setResult(null);
    setCopied(false);

    setTimeout(() => {
      const data = generateSmartData();
      setResult(data);
      saveToHistory(data);
      setLoading(false);
      setActiveTab('preview');
    }, 1200);
  };

  const handleCopy = (textData) => {
    const target = textData || result;
    if (!target) return;
    const textToCopy = `${target.title}\n\n${target.description}\n\nПреимущества:\n- ${target.bullets.join('\n- ')}`;
    navigator.clipboard.writeText(textToCopy);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="min-h-screen bg-slate-900 text-slate-100 py-10 px-4 sm:px-6 lg:px-8">
      <div className="max-w-6xl mx-auto space-y-10">
        
        {/* Шапка */}
        <div className="text-center">
          <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full bg-blue-500/10 border border-blue-500/20 text-blue-400 mb-4">
            <Sparkles className="w-4 h-4" />
            <span className="text-sm font-medium">React • Tailwind • Global E-Commerce AI</span>
          </div>
          <h1 className="text-4xl font-extrabold tracking-tight sm:text-5xl bg-gradient-to-r from-blue-400 via-indigo-300 to-purple-400 bg-clip-text text-transparent">
            Генератор SEO-карточек для маркетплейсов
          </h1>
          <p className="mt-3 text-base text-slate-400 max-w-2xl mx-auto">
            Интеллектуальное создание продающих описаний с оптимизацией под поисковые алгоритмы Amazon, Shopify и Etsy.
          </p>
        </div>

        {/* Сетка */}
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-8 items-start">
          
          {/* ЛЕВАЯ КОЛОНКА: Форма */}
          <div className="lg:col-span-5 bg-slate-800/50 border border-slate-700/50 rounded-2xl p-6 backdrop-blur-xl shadow-xl space-y-6">
            
            <div>
              <label className="block text-xs font-semibold uppercase tracking-wider text-slate-400 mb-2">
                Выберите платформу
              </label>
              <div className="grid grid-cols-3 gap-2">
                {[
                  { id: 'amazon', label: 'Amazon' },
                  { id: 'shopify', label: 'Shopify' },
                  { id: 'etsy', label: 'Etsy' },
                ].map((item) => (
                  <button
                    key={item.id}
                    type="button"
                    onClick={() => setMarketplace(item.id)}
                    className={`py-2 px-3 rounded-xl border text-sm font-medium transition flex items-center justify-center gap-1.5 ${
                      marketplace === item.id
                        ? item.id === 'amazon' ? 'bg-amber-600/20 border-amber-500 text-amber-300' : item.id === 'shopify' ? 'bg-emerald-600/20 border-emerald-500 text-emerald-300' : 'bg-orange-600/20 border-orange-500 text-orange-300'
                        : 'bg-slate-900/50 border-slate-700 text-slate-400 hover:border-slate-600'
                    }`}
                  >
                    <ShoppingBag className="w-3.5 h-3.5" />
                    <span>{item.label}</span>
                  </button>
                ))}
              </div>
            </div>

            <form onSubmit={handleGenerate} className="space-y-4">
              <div>
                <label className="block text-sm font-medium text-slate-300 mb-1">
                  Название товара <span className="text-red-400">*</span>
                </label>
                <input
                  type="text"
                  required
                  placeholder="Например: Wireless Noise-Canceling Headphones"
                  value={productName}
                  onChange={(e) => setProductName(e.target.value)}
                  className="w-full px-4 py-2.5 bg-slate-900/80 border border-slate-700 rounded-xl focus:outline-none focus:border-blue-500 transition text-slate-100 placeholder-slate-500 text-sm"
                />
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-sm font-medium text-slate-300 mb-1">
                    Цена ($ / €)
                  </label>
                  <input
                    type="text"
                    placeholder="49.99"
                    value={price}
                    onChange={(e) => setPrice(e.target.value)}
                    className="w-full px-4 py-2.5 bg-slate-900/80 border border-slate-700 rounded-xl focus:outline-none focus:border-blue-500 transition text-slate-100 text-sm"
                  />
                </div>
                <div>
                  <label className="block text-sm font-medium text-slate-300 mb-1">
                    Тон описания
                  </label>
                  <select
                    value={tone}
                    onChange={(e) => setTone(e.target.value)}
                    className="w-full px-4 py-2.5 bg-slate-900/80 border border-slate-700 rounded-xl focus:outline-none focus:border-blue-500 transition text-slate-200 text-sm"
                  >
                    <option value="selling">🔥 Продающий</option>
                    <option value="professional">👔 Строгий</option>
                    <option value="emotional">✨ Эмоциональный</option>
                  </select>
                </div>
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-300 mb-1">
                  Ключевые слова (через запятую)
                </label>
                <textarea
                  rows="3"
                  placeholder="bluetooth 5.3, deep bass, 30h battery life, ergonomic"
                  value={keywords}
                  onChange={(e) => setKeywords(e.target.value)}
                  className="w-full px-4 py-2.5 bg-slate-900/80 border border-slate-700 rounded-xl focus:outline-none focus:border-blue-500 transition resize-none text-slate-100 placeholder-slate-500 text-sm"
                />
              </div>

              <div className="pt-2">
                <button
                  type="submit"
                  disabled={loading || !productName}
                  className="w-full py-3 px-4 bg-gradient-to-r from-blue-600 via-indigo-600 to-purple-600 hover:from-blue-500 hover:to-purple-500 disabled:opacity-50 disabled:cursor-not-allowed rounded-xl font-medium shadow-lg shadow-indigo-500/25 flex items-center justify-center gap-2 transition duration-200"
                >
                  {loading ? (
                    <>
                      <RefreshCw className="w-5 h-5 animate-spin" />
                      <span>Нейросеть генерирует...</span>
                    </>
                  ) : (
                    <>
                      <Sparkles className="w-5 h-5" />
                      <span>Создать карточку</span>
                    </>
                  )}
                </button>
              </div>
            </form>
          </div>

          {/* ПРАВАЯ КОЛОНКА: Вывод результата */}
          <div className="lg:col-span-7 space-y-6">
            
            <div className="bg-slate-800/30 border border-slate-700/50 rounded-2xl p-6 backdrop-blur-xl min-h-[460px] flex flex-col justify-between shadow-xl">
              {loading ? (
                <div className="flex-1 flex flex-col items-center justify-center text-center p-8 space-y-4">
                  <div className="w-16 h-16 rounded-full bg-indigo-500/10 border border-indigo-500/20 flex items-center justify-center text-indigo-400 animate-pulse">
                    <Sparkles className="w-8 h-8 animate-spin" />
                  </div>
                  <div>
                    <h3 className="text-lg font-medium text-slate-200">Создаем визуал и тексты для {marketplace.toUpperCase()}...</h3>
                    <p className="text-sm text-slate-400 max-w-sm mt-1">
                      Оптимизируем структуру под международные стандарты e-commerce.
                    </p>
                  </div>
                </div>
              ) : result ? (
                <div className="space-y-6">
                  
                  <div className="flex flex-wrap items-center justify-between gap-4 border-b border-slate-700/60 pb-4">
                    <div className="flex bg-slate-900/80 p-1 rounded-xl border border-slate-700/60">
                      <button
                        onClick={() => setActiveTab('preview')}
                        className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition ${
                          activeTab === 'preview' ? 'bg-indigo-600 text-white shadow' : 'text-slate-400 hover:text-slate-200'
                        }`}
                      >
                        <Eye className="w-3.5 h-3.5" />
                        <span>Визуальное превью</span>
                      </button>
                      <button
                        onClick={() => setActiveTab('text')}
                        className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition ${
                          activeTab === 'text' ? 'bg-indigo-600 text-white shadow' : 'text-slate-400 hover:text-slate-200'
                        }`}
                      >
                        <FileText className="w-3.5 h-3.5" />
                        <span>Текст для копирования</span>
                      </button>
                    </div>

                    <button
                      onClick={() => handleCopy()}
                      className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-slate-700/80 hover:bg-slate-700 text-xs font-medium transition text-white"
                    >
                      {copied ? (
                        <>
                          <Check className="w-3.5 h-3.5 text-emerald-300" />
                          <span className="text-emerald-300">Скопировано!</span>
                        </>
                      ) : (
                        <>
                          <Copy className="w-3.5 h-3.5" />
                          <span>Скопировать текст</span>
                        </>
                      )}
                    </button>
                  </div>

                  {activeTab === 'preview' ? (
                    <div className="bg-slate-900/90 border border-slate-700/80 rounded-2xl p-5 shadow-inner space-y-6 animate-fadeIn">
                      <div className="flex items-center justify-between">
                        <span className="px-2.5 py-1 rounded text-xs font-black tracking-wider bg-indigo-500/20 text-indigo-400 border border-indigo-500/30">
                          {result.marketplace} GLOBAL STORE
                        </span>
                        <div className="flex items-center gap-1 text-amber-400 text-sm font-bold">
                          <Star className="w-4 h-4 fill-amber-400" />
                          <span>4.9</span>
                          <span className="text-slate-500 font-normal text-xs">(142 reviews)</span>
                        </div>
                      </div>

                      <div className="grid grid-cols-1 sm:grid-cols-12 gap-6 items-start">
                        <div className="sm:col-span-5 aspect-square bg-gradient-to-br from-slate-800 to-slate-800/40 border border-slate-700/60 rounded-xl flex flex-col items-center justify-center p-4 text-center relative group">
                          <ShoppingBag className="w-16 h-16 text-slate-600 mb-2 group-hover:scale-110 transition duration-300" />
                          <span className="text-xs text-slate-500 font-medium">Product Image</span>
                          <button className="absolute top-3 right-3 p-2 rounded-full bg-slate-900/60 text-slate-400 hover:text-red-400 transition">
                            <Heart className="w-4 h-4" />
                          </button>
                        </div>

                        <div className="sm:col-span-7 space-y-4">
                          <h2 className="text-base font-bold text-white leading-snug">
                            {result.title}
                          </h2>

                          <div className="flex items-baseline gap-2">
                            <span className="text-2xl font-black text-emerald-400">{result.currency}{result.price}</span>
                            <span className="text-sm text-slate-500 line-through">{result.currency}69.99</span>
                          </div>

                          <div className="space-y-1.5">
                            <span className="text-xs font-semibold text-slate-400 uppercase">Highlights:</span>
                            <ul className="space-y-1 text-xs text-slate-300">
                              {result.bullets.slice(0, 3).map((b, i) => (
                                <li key={i} className="flex items-center gap-1.5">
                                  <span className="w-1.5 h-1.5 rounded-full bg-indigo-400"></span>
                                  <span className="truncate">{b}</span>
                                </li>
                              ))}
                            </ul>
                          </div>

                          <button className="w-full py-2.5 px-4 bg-indigo-600 hover:bg-indigo-500 text-white rounded-xl text-sm font-semibold shadow-lg shadow-indigo-600/30 flex items-center justify-center gap-2 transition">
                            <ShoppingCart className="w-4 h-4" />
                            <span>Add to Cart</span>
                          </button>
                        </div>
                      </div>

                      <div className="border-t border-slate-800 pt-4 space-y-2">
                        <h4 className="text-xs font-bold uppercase tracking-wider text-slate-400">Product Description:</h4>
                        <p className="text-xs text-slate-300 leading-relaxed bg-slate-800/40 p-3.5 rounded-xl border border-slate-800/80">
                          {result.description}
                        </p>
                      </div>
                    </div>
                  ) : (
                    <div className="space-y-5 animate-fadeIn">
                      <div className="space-y-1">
                        <span className="text-xs font-semibold uppercase tracking-wider text-indigo-400">SEO-Название (Title)</span>
                        <div className="p-3.5 bg-slate-900/80 border border-slate-700/80 rounded-xl text-slate-100 font-medium text-sm">
                          {result.title}
                        </div>
                      </div>

                      <div className="space-y-1">
                        <span className="text-xs font-semibold uppercase tracking-wider text-indigo-400">Продающее описание</span>
                        <div className="p-3.5 bg-slate-900/80 border border-slate-700/80 rounded-xl text-slate-300 text-sm leading-relaxed whitespace-pre-wrap">
                          {result.description}
                        </div>
                      </div>

                      <div className="space-y-1">
                        <span className="text-xs font-semibold uppercase tracking-wider text-indigo-400">Преимущества (Bullet Points)</span>
                        <ul className="p-3.5 bg-slate-900/80 border border-slate-700/80 rounded-xl space-y-2 text-sm text-slate-300">
                          {result.bullets.map((bullet, index) => (
                            <li key={index} className="flex items-start gap-2">
                              <span className="text-indigo-400 font-bold">•</span>
                              <span>{bullet}</span>
                            </li>
                          ))}
                        </ul>
                      </div>
                    </div>
                  )}

                </div>
              ) : (
                <div className="flex-1 flex flex-col items-center justify-center text-center p-8 border-2 border-dashed border-slate-700/60 rounded-xl">
                  <div className="w-12 h-12 rounded-full bg-slate-800 flex items-center justify-center text-slate-500 mb-3">
                    <Sparkles className="w-6 h-6" />
                  </div>
                  <h3 className="text-base font-medium text-slate-300">Здесь появится карточка товара</h3>
                  <p className="text-sm text-slate-500 max-w-xs mt-1">
                    Заполните данные слева и нажмите кнопку генерации.
                  </p>
                </div>
              )}
            </div>

            {/* История */}
            {history.length > 0 && (
              <div className="bg-slate-800/20 border border-slate-700/40 rounded-2xl p-5 backdrop-blur-xl">
                <div className="flex items-center justify-between mb-4">
                  <div className="flex items-center gap-2 text-sm font-medium text-slate-300">
                    <History className="w-4 h-4 text-slate-400" />
                    <span>Недавние генерации</span>
                  </div>
                  <button onClick={clearHistory} className="text-xs text-slate-500 hover:text-red-400 flex items-center gap-1 transition">
                    <Trash2 className="w-3.5 h-3.5" />
                    <span>Очистить</span>
                  </button>
                </div>

                <div className="space-y-2">
                  {history.map((item) => (
                    <div
                      key={item.id}
                      onClick={() => { setResult(item); setActiveTab('preview'); }}
                      className="p-3 bg-slate-900/50 hover:bg-slate-900 border border-slate-700/50 rounded-xl cursor-pointer transition flex items-center justify-between group"
                    >
                      <div className="truncate pr-4">
                        <span className="text-xs font-bold text-indigo-400 mr-2">[{item.marketplace}]</span>
                        <span className="text-sm text-slate-300 group-hover:text-white transition">{item.productName}</span>
                      </div>
                      <span className="text-xs text-slate-500 group-hover:text-blue-400 transition whitespace-nowrap">Открыть →</span>
                    </div>
                  ))}
                </div>
              </div>
            )}

          </div>

        </div>

      </div>
    </div>
  );
}
