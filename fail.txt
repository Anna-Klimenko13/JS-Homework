import React, { useState } from 'react';

function App() {
  const [step, setStep] = useState('welcome');
  
  const [gender, setGender] = useState('male');
  const [age, setAge] = useState('');
  const [height, setHeight] = useState('');
  const [weight, setWeight] = useState('');

  // Состояние для хранения результатов
  const [results, setResults] = useState(null);

  // Функция расчета калорий и БЖУ
  const calculateNutrition = () => {
    // 1. Проверка: все ли поля заполнены
    if (!age || !height || !weight) {
      alert('Пожалуйста, заполните все параметры (Возраст, Рост, Вес)!');
      return;
    }

    const w = parseFloat(weight);
    const h = parseFloat(height);
    const a = parseFloat(age);

    // 2. Базовый обмен веществ (формула Миффлина-Сан Жеора)
    let bmr = 0;
    if (gender === 'male') {
      bmr = (10 * w) + (6.25 * h) - (5 * a) + 5;
    } else {
      bmr = (10 * w) + (6.25 * h) - (5 * a) - 161;
    }

    // 3. Добавляем коэффициент легкой активности (1.375)
    const tdee = Math.round(bmr * 1.375);

    // 4. Расчет БЖУ (Белки: 30%, Жиры: 30%, Углеводы: 40%)
    // 1г белка/углеводов = 4 ккал, 1г жира = 9 ккал
    const protein = Math.round((tdee * 0.3) / 4);
    const fat = Math.round((tdee * 0.3) / 9);
    const carbs = Math.round((tdee * 0.4) / 4);

    // Сохраняем результаты и переключаем экран
    setResults({ calories: tdee, protein, fat, carbs });
    setStep('result');
  };

  // Функция для возврата назад
  const resetForm = () => {
    setAge('');
    setHeight('');
    setWeight('');
    setStep('form');
  };

  return (
    <div className="min-h-screen flex flex-col items-center justify-center p-6">
      
      {/* --- ЭКРАН 1: СТАРТОВЫЙ --- */}
      {step === 'welcome' && (
        <div className="text-center flex flex-col items-center animate-fade-in">
          <h1 className="text-5xl font-bold text-white mb-3">
            Nutri<span className="text-primary">Q</span>
          </h1>
          <p className="text-textGray max-w-sm mb-10">
            Умный AI-расчет калорий и генератор персонального меню для достижения вашей цели.
          </p>
          <button 
            onClick={() => setStep('form')}
            className="bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 px-10 rounded-full transition-colors shadow-[0_0_20px_rgba(0,255,102,0.3)]"
          >
            Рассчитать норму
          </button>
        </div>
      )}

      {/* --- ЭКРАН 2: ФОРМА ПАРАМЕТРОВ --- */}
      {step === 'form' && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in">
          <h2 className="text-2xl font-bold text-white mb-6 text-center">Ваши параметры</h2>
          
          <div className="flex bg-dark rounded-xl p-1 mb-6">
            <button 
              onClick={() => setGender('male')}
              className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'male' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}
            >
              Мужчина
            </button>
            <button 
              onClick={() => setGender('female')}
              className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'female' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}
            >
              Женщина
            </button>
          </div>

          <div className="space-y-4">
            <div>
              <label className="block text-textGray text-sm font-medium mb-2">Возраст (лет)</label>
              <input 
                type="number" 
                value={age} 
                onChange={(e) => setAge(e.target.value)}
                placeholder="Например: 25" 
                className="w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors"
              />
            </div>
            
            <div className="flex gap-4">
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">Рост (см)</label>
                <input 
                  type="number" 
                  value={height} 
                  onChange={(e) => setHeight(e.target.value)}
                  placeholder="175" 
                  className="w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors"
                />
              </div>
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">Вес (кг)</label>
                <input 
                  type="number" 
                  value={weight} 
                  onChange={(e) => setWeight(e.target.value)}
                  placeholder="70" 
                  className="w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors"
                />
              </div>
            </div>
          </div>

          <button 
            onClick={calculateNutrition}
            className="w-full mt-8 bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 rounded-xl transition-colors"
          >
            Рассчитать
          </button>
        </div>
      )}

      {/* --- ЭКРАН 3: РЕЗУЛЬТАТЫ --- */}
      {step === 'result' && results && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in text-center">
          <h2 className="text-2xl font-bold text-white mb-2">Ваша норма</h2>
          <p className="text-textGray mb-8">С учетом базовой активности</p>
          
          {/* Главная цифра калорий */}
          <div className="mb-10">
            <span className="text-7xl font-black text-primary drop-shadow-[0_0_15px_rgba(0,255,102,0.3)]">
              {results.calories}
            </span>
            <span className="text-xl text-textGray ml-2 block mt-2">ккал / день</span>
          </div>

          {/* Плашки БЖУ */}
          <div className="flex justify-between gap-4 mb-8">
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">Белки</div>
              <div className="text-white font-bold text-xl">{results.protein}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">Жиры</div>
              <div className="text-white font-bold text-xl">{results.fat}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">Углеводы</div>
              <div className="text-white font-bold text-xl">{results.carbs}г</div>
            </div>
          </div>

          <button 
            onClick={resetForm}
            className="w-full bg-transparent border-2 border-primary text-primary hover:bg-primary hover:text-dark font-bold text-lg py-4 rounded-xl transition-all"
          >
            Рассчитать заново
          </button>
        </div>
      )}

    </div>
  );
}

export default App;
