import React, { useState } from 'react';

function App() {
  // Добавили состояние 'loading'
  const [step, setStep] = useState('welcome');
  
  const [gender, setGender] = useState('male');
  const [age, setAge] = useState('');
  const [height, setHeight] = useState('');
  const [weight, setWeight] = useState('');
  const [activity, setActivity] = useState('1.375');
  
  const [goal, setGoal] = useState('maintain');
  // НОВОЕ: Интенсивность (процент)
  const [intensity, setIntensity] = useState(15);
  
  const [results, setResults] = useState(null);

  const calculateNutrition = () => {
    if (!age || !height || !weight) {
      alert('Пожалуйста, заполните все параметры (Возраст, Рост, Вес)!');
      return;
    }

    // Включаем экран загрузки
    setStep('loading');

    // Имитируем работу AI на 2 секунды
    setTimeout(() => {
      const w = parseFloat(weight);
      const h = parseFloat(height);
      const a = parseFloat(age);

      // 1. Базовый обмен веществ (BMR)
      let bmr = (10 * w) + (6.25 * h) - (5 * a) + (gender === 'male' ? 5 : -161);
      let targetCalories = Math.round(bmr * parseFloat(activity));

      // 2. Корректировка по цели и интенсивности
      if (goal === 'lose') targetCalories = Math.round(targetCalories * (1 - intensity / 100));
      if (goal === 'gain') targetCalories = Math.round(targetCalories * (1 + intensity / 100));

      // 3. Расчет БЖУ
      const protein = Math.round((targetCalories * 0.3) / 4);
      const fat = Math.round((targetCalories * 0.3) / 9);
      const carbs = Math.round((targetCalories * 0.4) / 4);

      // 4. НОВОЕ: Расчет воды (35 мл на 1 кг веса)
      const water = (w * 0.035).toFixed(1);

      setResults({ calories: targetCalories, protein, fat, carbs, water, intensityApplied: intensity });
      
      // Переключаем на результаты
      setStep('result');
    }, 2000);
  };

  const resetForm = () => {
    setStep('form');
  };

  const blockInvalidChars = (e) => {
    if (e.key === '-' || e.key === 'e' || e.key === 'E') e.preventDefault();
  };

  const inputStyles = "w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors [appearance:textfield] [&::-webkit-outer-spin-button]:appearance-none [&::-webkit-inner-spin-button]:appearance-none";

  return (
    <div className="min-h-screen flex flex-col items-center justify-center p-6 pb-12">
      
      {/* --- ЭКРАН 1: СТАРТОВЫЙ --- */}
      {step === 'welcome' && (
        <div className="text-center flex flex-col items-center animate-fade-in">
          <h1 className="text-5xl font-bold text-white mb-3">Nutri<span className="text-primary">Q</span></h1>
          <p className="text-textGray max-w-sm mb-10">Умный AI-расчет калорий и генератор персонального меню.</p>
          <button onClick={() => setStep('form')} className="bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 px-10 rounded-full transition-colors shadow-[0_0_20px_rgba(0,255,102,0.3)]">
            Рассчитать норму
          </button>
        </div>
      )}

      {/* --- ЭКРАН 2: ФОРМА ПАРАМЕТРОВ --- */}
      {step === 'form' && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in">
          <h2 className="text-2xl font-bold text-white mb-6 text-center">Ваши параметры</h2>
          
          <div className="flex bg-dark rounded-xl p-1 mb-6">
            <button onClick={() => setGender('male')} className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'male' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}>Мужчина</button>
            <button onClick={() => setGender('female')} className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'female' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}>Женщина</button>
          </div>

          <div className="space-y-4">
            <div>
              <label className="block text-textGray text-sm font-medium mb-2">Возраст (лет)</label>
              <input type="number" min="1" value={age} onChange={(e) => setAge(e.target.value)} onKeyDown={blockInvalidChars} placeholder="25" className={inputStyles} />
            </div>
            
            <div className="flex gap-4">
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">Рост (см)</label>
                <input type="number" min="1" value={height} onChange={(e) => setHeight(e.target.value)} onKeyDown={blockInvalidChars} placeholder="175" className={inputStyles} />
              </div>
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">Вес (кг)</label>
                <input type="number" min="1" value={weight} onChange={(e) => setWeight(e.target.value)} onKeyDown={blockInvalidChars} placeholder="70" className={inputStyles} />
              </div>
            </div>

            <div className="pt-2">
              <label className="block text-textGray text-sm font-medium mb-2">Активность</label>
              <select value={activity} onChange={(e) => setActivity(e.target.value)} className="w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors cursor-pointer appearance-none">
                <option value="1.2">Минимум (сидячая работа, без спорта)</option>
                <option value="1.375">Слабая (тренировки 1-3 раза в неделю)</option>
                <option value="1.55">Средняя (тренировки 3-5 раз в неделю)</option>
                <option value="1.725">Высокая (тренировки 6-7 раз в неделю)</option>
              </select>
            </div>

            <div className="pt-2">
              <label className="block text-textGray text-sm font-medium mb-2">Ваша цель</label>
              <div className="flex flex-col gap-2 mb-4">
                <button onClick={() => setGoal('lose')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'lose' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>📉 Похудение</button>
                <button onClick={() => setGoal('maintain')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'maintain' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>⚖️ Поддержание веса</button>
                <button onClick={() => setGoal('gain')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'gain' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>💪 Набор массы</button>
              </div>

              {/* НОВОЕ: Выбор темпа появляется только если цель не "Поддержание" */}
              {goal !== 'maintain' && (
                <div className="animate-fade-in">
                  <label className="block text-textGray text-sm font-medium mb-2">Темп достижения цели</label>
                  <div className="flex gap-2">
                    <button onClick={() => setIntensity(10)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 10 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>Легкий (10%)</button>
                    <button onClick={() => setIntensity(15)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 15 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>Норма (15%)</button>
                    <button onClick={() => setIntensity(20)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 20 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>Быстрый (20%)</button>
                  </div>
                </div>
              )}
            </div>

          </div>

          <button onClick={calculateNutrition} className="w-full mt-8 bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 rounded-xl transition-colors">
            Рассчитать
          </button>
        </div>
      )}

      {/* --- ЭКРАН 3: ЗАГРУЗКА (Имитация AI) --- */}
      {step === 'loading' && (
        <div className="flex flex-col items-center justify-center animate-pulse">
          {/* Крутящийся спиннер */}
          <div className="w-16 h-16 border-4 border-dark border-t-primary rounded-full animate-spin mb-6"></div>
          <h2 className="text-2xl font-bold text-white mb-2">Анализ данных...</h2>
          <p className="text-textGray">NutriQ подбирает идеальные пропорции</p>
        </div>
      )}

      {/* --- ЭКРАН 4: РЕЗУЛЬТАТЫ --- */}
      {step === 'result' && results && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in text-center">
          <h2 className="text-2xl font-bold text-white mb-2">Ваша норма</h2>
          
          <div className="mb-8">
            <span className="text-7xl font-black text-primary drop-shadow-[0_0_15px_rgba(0,255,102,0.3)]">{results.calories}</span>
            <span className="text-xl text-textGray ml-2 block mt-2">ккал / день</span>
          </div>

          <div className="flex justify-between gap-4 mb-4">
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">Белки</div>
              <div className="text-white font-bold text-xl">{results.protein}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">Жиры</div>
              <div className="text-white font-bold text-xl">{results.fat}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">Углеводы</div>
              <div className="text-white font-bold text-xl">{results.carbs}г</div>
            </div>
          </div>

          {/* НОВОЕ: Карточка воды */}
          <div className="bg-blue-900/20 border border-blue-500/30 p-4 rounded-2xl mb-6 flex items-center justify-between">
            <div className="text-left">
              <div className="text-blue-300 font-bold">Норма воды</div>
              <div className="text-blue-200/70 text-sm">Поддерживает метаболизм</div>
            </div>
            <div className="text-2xl font-bold text-blue-400">~{results.water} л</div>
          </div>

          {/* НОВОЕ: Информационная сноска */}
          <p className="text-xs text-textGray/60 mb-6 px-4">
            * Расчет выполнен с учетом {goal === 'maintain' ? 'нулевого дефицита для поддержания веса' : `${goal === 'lose' ? 'дефицита' : 'профицита'} в ${results.intensityApplied}%`}. Формула Миффлина-Сан Жеора.
          </p>

          <button onClick={resetForm} className="w-full bg-transparent border-2 border-primary text-primary hover:bg-primary hover:text-dark font-bold text-lg py-4 rounded-xl transition-all">
            Изменить параметры
          </button>
        </div>
      )}

    </div>
  );
}

export default App;
