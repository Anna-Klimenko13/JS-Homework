import React, { useState } from 'react';

// --- СЛОВАРЬ ПЕРЕВОДОВ ---
const translations = {
  ru: {
    titleSub: "Умный AI-расчет калорий и генератор персонального меню.",
    btnStart: "Рассчитать норму",
    paramsTitle: "Ваши параметры",
    male: "Мужчина", female: "Женщина",
    age: "Возраст (лет)", height: "Рост (см)", weight: "Вес (кг)",
    activity: "Активность",
    act1: "Минимум (сидячая работа)", act2: "Слабая (1-3 тренировки)", act3: "Средняя (3-5 тренировок)", act4: "Высокая (6-7 тренировок)",
    goal: "Ваша цель",
    lose: "📉 Похудение", maintain: "⚖️ Поддержание веса", gain: "💪 Набор массы",
    pace: "Темп достижения цели",

    btnCalc: "Рассчитать",

    btnRecalc: "Изменить параметры", alert: "Пожалуйста, заполните все параметры!",
    
    // НОВЫЕ ПЕРЕВОДЫ ДЛЯ МЕНЮ
    btnGenMenu: "Сгенерировать меню",
    menuSetup: "Настройка плана питания",

  },
  en: {
    titleSub: "Smart AI calorie calculation and personal menu generator.",
    btnStart: "Calculate", paramsTitle: "Your Parameters",

    btnCreatePlan: "Create Plan",

    tabList: "🛒 Grocery List",

    breakfast: "Breakfast", lunch: "Lunch", dinner: "Dinner", snack: "Snack",
    backToNorm: "Back to Target"
  }
};

function App() {
  const [lang, setLang] = useState('ru');
  const t = translations[lang];

  const [step, setStep] = useState('welcome');
  const [gender, setGender] = useState('male');
  const [age, setAge] = useState('');
  const [height, setHeight] = useState('');
  const [weight, setWeight] = useState('');
  const [activity, setActivity] = useState('1.375');
  const [goal, setGoal] = useState('maintain');
  const [intensity, setIntensity] = useState(15);
  const [results, setResults] = useState(null);

  // Состояния для меню
  const [days, setDays] = useState(3);
  const [activeTab, setActiveTab] = useState('menu'); // 'menu' или 'list'

  const toggleLanguage = () => setLang(lang === 'ru' ? 'en' : 'ru');

  const calculateNutrition = () => {
    if (!age || !height || !weight) return alert(t.alert);
    setStep('loading');
    setTimeout(() => {
      const w = parseFloat(weight);
      const h = parseFloat(height);
      const a = parseFloat(age);
      let bmr = (10 * w) + (6.25 * h) - (5 * a) + (gender === 'male' ? 5 : -161);
      let targetCalories = Math.round(bmr * parseFloat(activity));
      if (goal === 'lose') targetCalories = Math.round(targetCalories * (1 - intensity / 100));
      if (goal === 'gain') targetCalories = Math.round(targetCalories * (1 + intensity / 100));
      const protein = Math.round((targetCalories * 0.3) / 4);
      const fat = Math.round((targetCalories * 0.3) / 9);
      const carbs = Math.round((targetCalories * 0.4) / 4);
      const water = (w * 0.035).toFixed(1);
      setResults({ calories: targetCalories, protein, fat, carbs, water, intensityApplied: intensity });
      setStep('result');
    }, 2000);
  };

  const blockInvalidChars = (e) => {
    if (e.key === '-' || e.key === 'e' || e.key === 'E') e.preventDefault();
  };

  const inputStyles = "w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors [appearance:textfield] [&::-webkit-outer-spin-button]:appearance-none [&::-webkit-inner-spin-button]:appearance-none";

  return (
    <div className="min-h-screen relative flex flex-col items-center justify-center p-6 pb-12">
      
      <button onClick={toggleLanguage} className="absolute top-6 right-6 bg-card border border-gray-700 text-white font-bold py-2 px-4 rounded-full hover:border-primary hover:text-primary transition-all">
        {lang === 'ru' ? 'EN' : 'RU'}
      </button>

      {/* --- 1. СТАРТОВЫЙ --- */}
      {step === 'welcome' && (
        <div className="text-center flex flex-col items-center animate-fade-in">
          <h1 className="text-5xl font-bold text-white mb-3">Nutri<span className="text-primary">Q</span></h1>
          <p className="text-textGray max-w-sm mb-10">{t.titleSub}</p>
          <button onClick={() => setStep('form')} className="bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 px-10 rounded-full transition-colors shadow-[0_0_20px_rgba(0,255,102,0.3)]">
            {t.btnStart}
          </button>
        </div>
      )}

      {/* --- 2. ФОРМА --- */}
      {step === 'form' && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in">
          <h2 className="text-2xl font-bold text-white mb-6 text-center">{t.paramsTitle}</h2>
          
          <div className="flex bg-dark rounded-xl p-1 mb-6">
            <button onClick={() => setGender('male')} className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'male' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}>{t.male}</button>
            <button onClick={() => setGender('female')} className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'female' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}>{t.female}</button>
          </div>

          <div className="space-y-4">
            <div>
              <label className="block text-textGray text-sm font-medium mb-2">{t.age}</label>
              <input type="number" min="1" value={age} onChange={(e) => setAge(e.target.value)} onKeyDown={blockInvalidChars} placeholder="25" className={inputStyles} />
            </div>
            <div className="flex gap-4">
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">{t.height}</label>
                <input type="number" min="1" value={height} onChange={(e) => setHeight(e.target.value)} onKeyDown={blockInvalidChars} placeholder="175" className={inputStyles} />
              </div>
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">{t.weight}</label>
                <input type="number" min="1" value={weight} onChange={(e) => setWeight(e.target.value)} onKeyDown={blockInvalidChars} placeholder="70" className={inputStyles} />
              </div>
            </div>

            <div className="pt-2">
              <label className="block text-textGray text-sm font-medium mb-2">{t.activity}</label>
              <select value={activity} onChange={(e) => setActivity(e.target.value)} className={inputStyles}>
                <option value="1.2">{t.act1}</option>
                <option value="1.375">{t.act2}</option>
                <option value="1.55">{t.act3}</option>
                <option value="1.725">{t.act4}</option>
              </select>
            </div>

            <div className="pt-2">
              <label className="block text-textGray text-sm font-medium mb-2">{t.goal}</label>
              <div className="flex flex-col gap-2 mb-4">
                <button onClick={() => setGoal('lose')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'lose' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>{t.lose}</button>
                <button onClick={() => setGoal('maintain')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'maintain' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>{t.maintain}</button>
                <button onClick={() => setGoal('gain')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'gain' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>{t.gain}</button>
              </div>

              {goal !== 'maintain' && (
                <div className="animate-fade-in">
                  <label className="block text-textGray text-sm font-medium mb-2">{t.pace}</label>
                  <div className="flex gap-2">
                    <button onClick={() => setIntensity(10)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 10 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>{t.pace1}</button>
                    <button onClick={() => setIntensity(15)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 15 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>{t.pace2}</button>
                    <button onClick={() => setIntensity(20)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 20 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>{t.pace3}</button>
                  </div>
                </div>
              )}
            </div>
          </div>
          <button onClick={calculateNutrition} className="w-full mt-8 bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 rounded-xl transition-colors">{t.btnCalc}</button>
        </div>
      )}

      {/* --- 3. ПЛАВНАЯ ЗАГРУЗКА --- */}
      {step === 'loading' && (
        <div className="flex flex-col items-center justify-center">
          <div className="relative flex justify-center items-center mb-10 mt-10 w-48 h-48">
            {/* Глубокая плавная пульсация вместо резкого радара */}
            <div className="absolute w-full h-full bg-primary/10 rounded-full animate-[pulse_3s_ease-in-out_infinite]"></div>
            <div className="absolute w-32 h-32 bg-primary/20 rounded-full animate-[pulse_2s_ease-in-out_infinite]"></div>
            <h1 className="relative text-5xl font-bold text-white z-10 drop-shadow-[0_0_20px_rgba(0,255,102,0.8)]">
              Nutri<span className="text-primary">Q</span>
            </h1>
          </div>
          <h2 className="text-2xl font-bold text-white mb-2">{t.loading}</h2>
          <p className="text-textGray">{t.loadingSub}</p>
        </div>
      )}

      {/* --- 4. РЕЗУЛЬТАТЫ --- */}
      {step === 'result' && results && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in text-center">
          <h2 className="text-2xl font-bold text-white mb-2">{t.normTitle}</h2>
          
          <div className="mb-8 mt-6">
            <span className="text-7xl font-black text-primary drop-shadow-[0_0_15px_rgba(0,255,102,0.3)]">{results.calories}</span>
            <span className="text-xl text-textGray ml-2 block mt-2">{t.kcal}</span>
          </div>

          <div className="flex justify-between gap-4 mb-6">
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">{t.protein}</div>
              <div className="text-white font-bold text-xl">{results.protein}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">{t.fat}</div>
              <div className="text-white font-bold text-xl">{results.fat}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">{t.carbs}</div>
              <div className="text-white font-bold text-xl">{results.carbs}г</div>
            </div>
          </div>

          <div className="bg-blue-900/20 border border-blue-500/30 p-4 rounded-2xl mb-8 flex items-center justify-between">
            <div className="text-left">
              <div className="text-blue-300 font-bold">{t.waterTitle}</div>
              <div className="text-blue-200/70 text-sm">{t.waterSub}</div>
            </div>
            <div className="text-2xl font-bold text-blue-400">~{results.water} л</div>
          </div>

          {/* НОВАЯ КНОПКА: Переход к генерации меню */}
          <button onClick={() => setStep('menu-setup')} className="w-full bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 rounded-xl transition-colors mb-4 shadow-[0_0_15px_rgba(0,255,102,0.2)]">
            {t.btnGenMenu}
          </button>
          
          <button onClick={() => setStep('form')} className="w-full bg-transparent text-textGray hover:text-white font-medium py-3 transition-all">
            {t.btnRecalc}
          </button>
        </div>
      )}

      {/* --- 5. НАСТРОЙКА МЕНЮ --- */}
      {step === 'menu-setup' && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in text-center">
          <h2 className="text-2xl font-bold text-white mb-6">{t.menuSetup}</h2>
          
          <label className="block text-textGray mb-4">{t.daysCount} <span className="text-white font-bold text-xl ml-2">{days}</span></label>
          
          <div className="flex justify-between gap-2 mb-10">
            {[1, 2, 3, 4, 5, 6, 7].map(num => (
              <button 
                key={num}
                onClick={() => setDays(num)}
                className={`w-12 h-12 rounded-full font-bold transition-all flex items-center justify-center ${days === num ? 'bg-primary text-dark shadow-[0_0_10px_rgba(0,255,102,0.5)]' : 'bg-dark text-textGray hover:text-white'}`}
              >
                {num}
              </button>
            ))}
          </div>

          <button onClick={() => setStep('menu-result')} className="w-full bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 rounded-xl transition-colors mb-4">
            {t.btnCreatePlan}
          </button>
          <button onClick={() => setStep('result')} className="w-full bg-transparent text-textGray hover:text-white transition-all py-2">
            {t.backToNorm}
          </button>
        </div>
      )}

      {/* --- 6. ИТОГОВОЕ МЕНЮ И СПИСОК ПОКУПОК --- */}
      {step === 'menu-result' && (
        <div className="bg-card w-full max-w-md p-6 rounded-3xl shadow-2xl animate-fade-in h-[80vh] flex flex-col">
          
          {/* Переключатель вкладок (Табы) */}
          <div className="flex bg-dark rounded-xl p-1 mb-6 shrink-0">
            <button onClick={() => setActiveTab('menu')} className={`flex-1 py-3 rounded-lg font-bold transition-all ${activeTab === 'menu' ? 'bg-primary text-dark' : 'text-textGray'}`}>{t.tabMenu}</button>
            <button onClick={() => setActiveTab('list')} className={`flex-1 py-3 rounded-lg font-bold transition-all ${activeTab === 'list' ? 'bg-primary text-dark' : 'text-textGray'}`}>{t.tabList}</button>
          </div>

          {/* Скроллируемая область контента */}
          <div className="flex-1 overflow-y-auto pr-2 custom-scrollbar">
            
            {/* Вкладка 1: МЕНЮ */}
            {activeTab === 'menu' && (
              <div className="space-y-6">
                {Array.from({ length: days }).map((_, i) => (
                  <div key={i} className="bg-dark p-5 rounded-2xl border border-gray-800">
                    <h3 className="text-primary font-bold text-lg mb-4 border-b border-gray-700 pb-2">{t.dayTitle} {i + 1}</h3>
                    
                    <div className="space-y-4">
                      <div>
                        <div className="text-xs text-textGray uppercase tracking-wider mb-1">{t.breakfast}</div>
                        <div className="text-white text-sm">{lang === 'ru' ? 'Овсяноблин с творожным сыром и слабосоленой семгой' : 'Oatmeal pancake with cream cheese and salmon'}</div>
                      </div>
                      <div>
                        <div className="text-xs text-textGray uppercase tracking-wider mb-1">{t.snack}</div>
                        <div className="text-white text-sm">{lang === 'ru' ? 'Горсть грецких орехов и зеленое яблоко' : 'Handful of walnuts and a green apple'}</div>
                      </div>
                      <div>
                        <div className="text-xs text-textGray uppercase tracking-wider mb-1">{t.lunch}</div>
                        <div className="text-white text-sm">{lang === 'ru' ? 'Куриное филе на гриле с киноа и свежими овощами' : 'Grilled chicken breast with quinoa and fresh veg'}</div>
                      </div>
                      <div>
                        <div className="text-xs text-textGray uppercase tracking-wider mb-1">{t.dinner}</div>
                        <div className="text-white text-sm">{lang === 'ru' ? 'Запеченный лосось со спаржей под лимонным соком' : 'Baked salmon with asparagus and lemon juice'}</div>
                      </div>
                    </div>
                  </div>
                ))}
              </div>
            )}

            {/* Вкладка 2: СПИСОК ПОКУПОК */}
            {activeTab === 'list' && (
              <div className="bg-dark p-5 rounded-2xl border border-gray-800">
                <p className="text-textGray text-sm mb-6 pb-4 border-b border-gray-700">
                  {lang === 'ru' ? 'Ингредиенты рассчитаны на' : 'Ingredients calculated for'} {days} {lang === 'ru' ? 'дней' : 'days'}
                </p>
                
                {/* Чекбоксы списка покупок */}
                <div className="space-y-3">
                  {['Овсяные хлопья / Oats', 'Творожный сыр / Cream cheese', 'Слабосоленая семга / Salted salmon', 'Грецкие орехи / Walnuts', 'Зеленые яблоки / Green apples', 'Куриное филе / Chicken breast', 'Киноа / Quinoa', 'Свежие огурцы и томаты / Cucumbers & tomatoes', 'Свежий лосось / Fresh salmon', 'Спаржа / Asparagus', 'Лимон / Lemon'].map((item, i) => (
                    <label key={i} className="flex items-center gap-3 cursor-pointer group">
                      <input type="checkbox" className="w-5 h-5 accent-primary bg-gray-700 border-gray-600 rounded cursor-pointer" />
                      <span className="text-white group-hover:text-primary transition-colors">
                        {lang === 'ru' ? item.split(' / ')[0] : item.split(' / ')[1]}
                      </span>
                    </label>
                  ))}
                </div>
              </div>
            )}
          </div>
          
          {/* Кнопка "Назад" в самом низу */}
          <button onClick={() => setStep('result')} className="w-full mt-4 bg-transparent border border-gray-700 text-textGray hover:text-white font-medium py-3 rounded-xl transition-all shrink-0">
            {t.backToNorm}
          </button>
        </div>
      )}

    </div>
  );
}

export default App;
