// --- СЛОВАРЬ ПЕРЕВОДОВ ---

  ru: {

  },
  en: {

    btnRecalc: "Change Parameters",
    alert: "Please fill in all parameters!"
  }

function App() {
  // Состояние языка (по умолчанию 'ru')
  const [lang, setLang] = useState('ru');
  const t = translations[lang];

  const [step, setStep] = useState('welcome');
  const [gender, setGender] = useState('male');
  const [age, setAge] = useState('');
  const [height, setHeight] = useState('');
  const [weight, setWeight] = useState('');
  const [activity, setActivity] = useState('1.375');
  const [goal, setGoal] = useState('maintain');
  const [intensity, setIntensity] = useState(15);
  const [results, setResults] = useState(null);

  const toggleLanguage = () => {
    setLang(lang === 'ru' ? 'en' : 'ru');
  };

  const calculateNutrition = () => {
    if (!age || !height || !weight) {
      alert(t.alert);
      return;
    }

    setStep('loading');

    setTimeout(() => {
      const w = parseFloat(weight);
      const h = parseFloat(height);
      const a = parseFloat(age);

      let bmr = (10 * w) + (6.25 * h) - (5 * a) + (gender === 'male' ? 5 : -161);
      let targetCalories = Math.round(bmr * parseFloat(activity));

      if (goal === 'lose') targetCalories = Math.round(targetCalories * (1 - intensity / 100));
      if (goal === 'gain') targetCalories = Math.round(targetCalories * (1 + intensity / 100));

      const protein = Math.round((targetCalories * 0.3) / 4);
      const fat = Math.round((targetCalories * 0.3) / 9);
      const carbs = Math.round((targetCalories * 0.4) / 4);
      const water = (w * 0.035).toFixed(1);

      setResults({ calories: targetCalories, protein, fat, carbs, water, intensityApplied: intensity });
      setStep('result');
    }, 2000);
  };

  const resetForm = () => {
    setStep('form');
  };

  const blockInvalidChars = (e) => {
    if (e.key === '-' || e.key === 'e' || e.key === 'E') e.preventDefault();
  };

  const inputStyles = "w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors [appearance:textfield] [&::-webkit-outer-spin-button]:appearance-none [&::-webkit-inner-spin-button]:appearance-none";

  return (
    <div className="min-h-screen relative flex flex-col items-center justify-center p-6 pb-12">
      
      {/* Кнопка переключения языка (фиксированная сверху справа) */}
      <button 
        onClick={toggleLanguage}
        className="absolute top-6 right-6 bg-card border border-gray-700 text-white font-bold py-2 px-4 rounded-full hover:border-primary hover:text-primary transition-all"
      >
        {lang === 'ru' ? 'EN' : 'RU'}
      </button>

      {/* --- ЭКРАН 1: СТАРТОВЫЙ --- */}
      {step === 'welcome' && (
        <div className="text-center flex flex-col items-center animate-fade-in">
          <h1 className="text-5xl font-bold text-white mb-3">Nutri<span className="text-primary">Q</span></h1>
          <p className="text-textGray max-w-sm mb-10">{t.titleSub}</p>
          <button onClick={() => setStep('form')} className="bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 px-10 rounded-full transition-colors shadow-[0_0_20px_rgba(0,255,102,0.3)]">
            {t.btnStart}
          </button>
        </div>
      )}

      {/* --- ЭКРАН 2: ФОРМА ПАРАМЕТРОВ --- */}
      {step === 'form' && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in">
          <h2 className="text-2xl font-bold text-white mb-6 text-center">{t.paramsTitle}</h2>
          
          <div className="flex bg-dark rounded-xl p-1 mb-6">
            <button onClick={() => setGender('male')} className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'male' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}>{t.male}</button>
            <button onClick={() => setGender('female')} className={`flex-1 py-2 rounded-lg font-bold transition-all ${gender === 'female' ? 'bg-primary text-dark shadow-md' : 'text-textGray hover:text-white'}`}>{t.female}</button>
          </div>

          <div className="space-y-4">
            <div>
              <label className="block text-textGray text-sm font-medium mb-2">{t.age}</label>
              <input type="number" min="1" value={age} onChange={(e) => setAge(e.target.value)} onKeyDown={blockInvalidChars} placeholder="25" className={inputStyles} />
            </div>
            
            <div className="flex gap-4">
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">{t.height}</label>
                <input type="number" min="1" value={height} onChange={(e) => setHeight(e.target.value)} onKeyDown={blockInvalidChars} placeholder="175" className={inputStyles} />
              </div>
              <div className="flex-1">
                <label className="block text-textGray text-sm font-medium mb-2">{t.weight}</label>
                <input type="number" min="1" value={weight} onChange={(e) => setWeight(e.target.value)} onKeyDown={blockInvalidChars} placeholder="70" className={inputStyles} />
              </div>
            </div>

            <div className="pt-2">
              <label className="block text-textGray text-sm font-medium mb-2">{t.activity}</label>
              <select value={activity} onChange={(e) => setActivity(e.target.value)} className="w-full bg-dark text-white p-4 rounded-xl outline-none border border-transparent focus:border-primary transition-colors cursor-pointer appearance-none">
                <option value="1.2">{t.act1}</option>
                <option value="1.375">{t.act2}</option>
                <option value="1.55">{t.act3}</option>
                <option value="1.725">{t.act4}</option>
              </select>
            </div>

            <div className="pt-2">
              <label className="block text-textGray text-sm font-medium mb-2">{t.goal}</label>
              <div className="flex flex-col gap-2 mb-4">
                <button onClick={() => setGoal('lose')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'lose' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>{t.lose}</button>
                <button onClick={() => setGoal('maintain')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'maintain' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>{t.maintain}</button>
                <button onClick={() => setGoal('gain')} className={`py-3 px-4 rounded-xl font-bold transition-all text-left border ${goal === 'gain' ? 'bg-dark border-primary text-primary' : 'bg-dark border-transparent text-textGray hover:text-white'}`}>{t.gain}</button>
              </div>

              {goal !== 'maintain' && (
                <div className="animate-fade-in">
                  <label className="block text-textGray text-sm font-medium mb-2">{t.pace}</label>
                  <div className="flex gap-2">
                    <button onClick={() => setIntensity(10)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 10 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>{t.pace1}</button>
                    <button onClick={() => setIntensity(15)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 15 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>{t.pace2}</button>
                    <button onClick={() => setIntensity(20)} className={`flex-1 py-2 text-sm rounded-lg font-bold transition-all ${intensity === 20 ? 'bg-primary/20 text-primary border-primary' : 'bg-dark border-transparent text-textGray'} border`}>{t.pace3}</button>
                  </div>
                </div>
              )}
            </div>

          </div>

          <button onClick={calculateNutrition} className="w-full mt-8 bg-primary hover:bg-primaryHover text-dark font-bold text-lg py-4 rounded-xl transition-colors">
            {t.btnCalc}
          </button>
        </div>
      )}

      {/* --- ЭКРАН 3: ПУЛЬСИРУЮЩАЯ ЗАГРУЗКА --- */}
      {step === 'loading' && (
        <div className="flex flex-col items-center justify-center">
          <div className="relative flex justify-center items-center mb-10 mt-10">
            {/* Анимация расходящихся волн (радар) */}
            <div className="absolute w-32 h-32 bg-primary/30 rounded-full animate-ping"></div>
            <div className="absolute w-24 h-24 bg-primary/50 rounded-full animate-pulse"></div>
            {/* Логотип по центру */}
            <h1 className="relative text-5xl font-bold text-white z-10 drop-shadow-[0_0_20px_rgba(0,255,102,1)]">
              Nutri<span className="text-primary">Q</span>
            </h1>
          </div>
          <h2 className="text-2xl font-bold text-white mb-2 animate-pulse">{t.loading}</h2>
          <p className="text-textGray">{t.loadingSub}</p>
        </div>
      )}

      {/* --- ЭКРАН 4: РЕЗУЛЬТАТЫ --- */}
      {step === 'result' && results && (
        <div className="bg-card w-full max-w-md p-8 rounded-3xl shadow-2xl animate-fade-in text-center">
          <h2 className="text-2xl font-bold text-white mb-2">{t.normTitle}</h2>
          
          <div className="mb-8 mt-6">
            <span className="text-7xl font-black text-primary drop-shadow-[0_0_15px_rgba(0,255,102,0.3)]">{results.calories}</span>
            <span className="text-xl text-textGray ml-2 block mt-2">{t.kcal}</span>
          </div>

          <div className="flex justify-between gap-4 mb-4">
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">{t.protein}</div>
              <div className="text-white font-bold text-xl">{results.protein}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">{t.fat}</div>
              <div className="text-white font-bold text-xl">{results.fat}г</div>
            </div>
            <div className="bg-dark flex-1 p-4 rounded-2xl border border-gray-800">
              <div className="text-textGray text-sm mb-1">{t.carbs}</div>
              <div className="text-white font-bold text-xl">{results.carbs}г</div>
            </div>
          </div>

          <div className="bg-blue-900/20 border border-blue-500/30 p-4 rounded-2xl mb-6 flex items-center justify-between">
            <div className="text-left">
              <div className="text-blue-300 font-bold">{t.waterTitle}</div>
              <div className="text-blue-200/70 text-sm">{t.waterSub}</div>
            </div>
            <div className="text-2xl font-bold text-blue-400">~{results.water} л</div>
          </div>

          <p className="text-xs text-textGray/60 mb-6 px-4">
            {t.noteBase} {goal === 'maintain' ? t.noteMaintain : ` ${results.intensityApplied}% ${goal === 'lose' ? t.noteDeficit : t.noteSurplus}.`}
          </p>

          <button onClick={resetForm} className="w-full bg-transparent border-2 border-primary text-primary hover:bg-primary hover:text-dark font-bold text-lg py-4 rounded-xl transition-all">
            {t.btnRecalc}
          </button>
        </div>
      )}

    </div>
  );
}

export default App;
